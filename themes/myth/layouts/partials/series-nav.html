{{- /* Prev / next navigation within the same series, ordered by verse or day. */ -}} {{- $s := .Params.series -}} {{- $siblings := where site.RegularPages "Params.series" $s -}} {{- $order := "Params.verse" -}} {{- if not .Params.verse }}{{ $order = "Params.day" }}{{ end }} {{- $sorted := sort $siblings $order "asc" -}} {{- $pos := 0 -}} {{- range $i, $p := $sorted -}} {{- if eq $p.Permalink $.Permalink }}{{ $pos = $i }}{{ end -}} {{- end -}} {{- $prev := index $sorted (sub $pos 1) -}} {{- $next := index $sorted (add $pos 1) -}} {{- if or $prev $next }}
{{- with $prev }} ← Previous {{ .Params.part | default .Title }} {{- else }}{{ end }} All {{ len $siblings }} {{- with $next }} Next → {{ .Params.part | default .Title }} {{- else }}{{ end }}
{{- end -}}
